    def crop_to_proggies(path):
        """Capture tight bounding box around all visible VB6 forms."""
        cur_windows = wd.snapshot()
        cur_forms = wd.find_vb_forms(cur_windows)
        if not cur_forms:
            return capture_cropped(main_win, str(path))
        # Tight bounding box around all visible VB6 forms
        x0 = min(w['x'] for w in cur_forms)
        y0 = min(w['y'] for w in cur_forms)
        x1 = max(w['x'] + w['w'] for w in cur_forms)
        y1 = max(w['y'] + w['h'] for w in cur_forms)
        pad = 4
        tight = {'x': max(x0 - pad, 0), 'y': max(y0 - pad, 0),
                 'w': min(x1 - x0 + 2 * pad, SCREEN_W - max(x0 - pad, 0)),
                 'h': min(y1 - y0 + 2 * pad, SCREEN_H - max(y0 - pad, 0))}
        return capture_cropped(tight, str(path))

=F7=
severity: MEDIUM
title: File handle leak from hunter trace stream — never closed
line: 19-20
description: `open(_hunter_log, 'a')` creates a file handle stored only as an argument to `hunter.trace()`. It is never closed explicitly nor assigned to a module-level variable to prevent GC. Per PEP 343, file handles should be managed via context managers or at minimum held in a named variable. While CPython's refcount GC would not collect it (the CallPrinter holds a reference), this is a resource leak that can cause issues with log rotation or filesystem unmount. Additionally, if `hunter.stop()` is ever called, the stream remains open with no reference to close it.
===CURRENT_CODE===
hunter.trace(stdlib=False, action=hunter.CallPrinter(
    stream=open(_hunter_log, 'a')))
===FIXED_CODE===
_hunter_stream = open(_hunter_log, 'a')
hunter.trace(stdlib=False, action=hunter.CallPrinter(
    stream=_hunter_stream))
# _hunter_stream held at module level; close via atexit if needed
