  Pythondef qga_cmd(self, cmd):
    self.sock.sendall(json.dumps(cmd).encode('utf-8') + b'\n')
    buffer = ""
    while True:
        try:
            data = self.sock.recv(4096).decode('utf-8')
            if not data:
                break
            buffer += data

            # Split buffer by newlines to handle multiple JSON objects
            if '\n' in buffer:
                lines = buffer.split('\n')
                # Retain the last element in the buffer (empty if ended cleanly, or partial line)
                buffer = lines.pop()

                for line in lines:
                    line = line.strip()
                    if not line:
                        continue
                    try:
                        resp = json.loads(line)
                        if "return" in resp or "error" in resp:
                            return resp
                    except json.JSONDecodeError:
                        # Skip malformed payloads instead of crashing
                        continue
        except socket.timeout:
            break
    return None

        written = resp.get("return", {}).get("count", 0)
        if written == 0:
            raise RuntimeError("QGA Write Error: 0 bytes written, potential stalling")

        offset += written

    return offset
