  Bash#!/usr/bin/env bash
set -uo pipefail

REPO="${1:?Usage: $0 <repo_root>}"
APPLIED=0
FAILED=0
SKIPPED=0
declare -a FIXES_APPLIED=()
declare -a FIXES_FAILED=()

apply_fix() {
    local desc="$1"
    local file="$2"
    local tmppy
    tmppy=$(mktemp /tmp/fix_XXXXXX.py)
    cat > "$tmppy"

    if [ ! -f "$file" ]; then
        echo "  SKIP: $desc -- file not found: $file"
        SKIPPED=$((SKIPPED+1))
        rm -f "$tmppy"
        return 0
    fi

    cp "$file" "${file}.fixbak"

    if python3 "$tmppy" "$file" 2>&1; then
        if python3 -m py_compile "$file" 2>/dev/null; then
            echo "  APPLIED: $desc"
            rm -f "${file}.fixbak" "$tmppy"
            APPLIED=$((APPLIED+1))
            FIXES_APPLIED+=("$desc")
            return 0
        else
            echo "  COMPILE-FAIL: $desc -- reverted"
            mv "${file}.fixbak" "$file"
            FAILED=$((FAILED+1))
            FIXES_FAILED+=("$desc (compile)")
            rm -f "$tmppy"
            return 1
        fi
    else
        echo "  PATCH-FAIL: $desc -- pattern not found, reverted"
        mv "${file}.fixbak" "$file"
        FAILED=$((FAILED+1))
        FIXES_FAILED+=("$desc (pattern)")
