def is_16bit_ne(exe_path):
    """Check if an executable is a 16-bit NE format (not supported by VB Decompiler)."""
    try:
        with open(exe_path, 'rb') as f:
            if f.read(2) != b'MZ': return False
            f.seek(0x3C)
            pe_off = struct.unpack('&lt;I', f.read(4))[0]
            f.seek(pe_off)
            return f.read(2) == b'NE'
    except Exception:
        return False
