        c2._poisoned = False

    if c2._poisoned and cmd != 'PING':
        return None

    try:
        os.remove(RES_FILE)
    except FileNotFoundError:
        pass
    except OSError:
        pass

    tmp_cmd = f'{CMD_FILE}.{os.getpid()}.tmp'
    with open(tmp_cmd, 'w') as f:
        f.write(cmd + '\n')
        f.flush()
        os.fsync(f.fileno())
    os.chown(tmp_cmd, 994, 1005)
    os.replace(tmp_cmd, CMD_FILE)

    sent_at = time.time()
    deadline = sent_at + timeout
    while time.time() < deadline:
        free_process()
        try:
            st = os.stat(RES_FILE)
            if st.st_mtime < sent_at:
                continue
            with open(RES_FILE) as f:
                data = f.read().strip().replace('\r', '')
            if not data:
                continue
            if cmd == 'PING' and ('PONG' in data or 'INJECTED' in data):
                c2._poisoned = False
            return data
        except FileNotFoundError:
            pass

    c2._poisoned = True
    try:
        os.remove(CMD_FILE)
    except OSError:
        pass
    try:
        os.remove(RES_FILE)
    except OSError:
        pass
    return None

  Pythondef get_menu_info(hwnd):
    """Get top-level menu command info. Returns (items, count) or (None, 0)."""
    menus = c2(f'ENUMMENUS {hwnd}', timeout=5)
    if not menus or menus in ('EMPTY', 'ERR: no menu'):
        return None, 0

    items = []
    seen = set()
    for line in menus.split('\n'):
        parts = line.split('|')
        if len(parts) != 3 or not parts[2].isdigit():
            continue
        caption = parts[1].replace('&', '').strip()
        cmd_id = int(parts[2])
        if not caption or cmd_id <= 0:
            continue
        key = (caption.lower(), cmd_id)
        if key in seen:
            continue
        seen.add(key)
        items.append({'caption': caption, 'command': cmd_id})

    return (items or None), len(items)
