def _xkey(hwnd, key):
    """Send a key via xdotool to a specific window. Skips if window not found.

    F6: Never sends to globally focused window — could hit c2host or
    the main app window outside the controlled flow.
    """
    xid = _find_x11_window(hwnd)
    if xid:
        subprocess.run(
            ['xdotool', 'windowactivate', str(xid)],
            env={'DISPLAY': DISPLAY}, timeout=3, capture_output=True)
        time.sleep(0.1)
        subprocess.run(
            ['xdotool', 'key', '--window', str(xid), key],
            env={'DISPLAY': DISPLAY}, timeout=3, capture_output=True)
    else:
        log.warning('_xkey: cannot find X11 window for hwnd=%d, '
                    'dropping key "%s" (would go to wrong window)', hwnd, key)

def _cycle_tabs(main_hwnd, step, out_dir, frames, max_tabs=6):
    """Cycle through tab pages via Ctrl+PageDown, screenshot each."""
    for i in range(max_tabs):
        _xkey(main_hwnd, 'ctrl+Next')  # Ctrl+PageDown
        time.sleep(TIMING['tab_delay'])  # F20: configurable
        safe = 'tab_%d' % (i + 1)
        shot_path = os.path.join(out_dir, '%02d_%s.bmp' % (step, safe))
        if c2_screenshot(main_hwnd, shot_path):
            if not frames or not _frames_identical(frames[-1], shot_path):
                frames.append(shot_path)
                log.info('Frame %d: tab %d', step, i + 1)
                step += 1
            else:
                os.remove(shot_path)
                log.info('Tab %d: duplicate, stopping tab cycle', i + 1)
                break
    return step

def _assemble_gif(exe_name, frames, out_dir):
    """Assemble BMP frames into animated GIF."""
    if len(frames) < 1:
        log.info('No frames to assemble')
        return

    try:
        from PIL import Image
    except ImportError:
        log.error('Pillow not installed, skipping GIF assembly')
        return

    gif_path = os.path.join(out_dir, exe_name + '.gif')
    pil_frames = []
    for f in frames:
        try:
            img = Image.open(f).convert('RGB')
            # Upscale small images
            if img.width < 400:
                scale = max(2, 400 // img.width)
                img = img.resize((img.width * scale, img.height * scale),
                                 Image.NEAREST)
            pil_frames.append(img)
        except Exception as e:
            log.warning('Cannot load %s: %s', f, e)

    if not pil_frames:
        return

    # F18: Normalize all frames to same size — scale to fit, not crop
    w, h = pil_frames[0].size
    normalized = []
    for img in pil_frames:
        if img.size != (w, h):
            # Scale to fit within target dimensions, preserving aspect ratio
            scale_factor = min(w / img.width, h / img.height)
            new_w = int(img.width * scale_factor)
            new_h = int(img.height * scale_factor)
            resample = (Image.LANCZOS if hasattr(Image, 'LANCZOS')
                        else Image.BICUBIC)
            img = img.resize((new_w, new_h), resample)
            # Center on gray background
            bg = Image.new('RGB', (w, h), (192, 192, 192))
            ox = (w - img.width) // 2
            oy = (h - img.height) // 2
            bg.paste(img, (ox, oy))
            normalized.append(bg)
        else:
            normalized.append(img)

    # Convert to palette with full 256 colors
    palettized = [img.quantize(colors=256, method=2) for img in normalized]
    palettized[0].save(
        gif_path, save_all=True, append_images=palettized[1:],
        duration=2000, loop=0
    )
    log.info('GIF: %s (%d frames, %d KB)',
             gif_path, len(normalized),
             os.path.getsize(gif_path) // 1024)


Note on F14 (Caption-Based Safety Model)
F14 is a design-level flaw — controls named Command1 with no descriptive caption bypass all string-matching safety checks. This is architecturally addressed by:

F1 fix: Positive-match invocation means only IDs cross-referenced against the nav graph are invoked. Unknown IDs are skipped entirely when safe_menu_ids is provided.
F2 fix: Phase 3 no longer falls back to clicking all unmatched controls.
F7 fix: Without a nav graph, no menu invocation or control clicking occurs.

A complete fix would require a capability-based safety model (sandboxing, network monitoring, or filesystem diffing per-invocation), which is beyond a code-level patch.
